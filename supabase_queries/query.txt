-- Function to check if a schema exists
CREATE OR REPLACE FUNCTION get_schema_exists(p_schema_name text)
RETURNS boolean AS $$
BEGIN
  RETURN EXISTS (
    SELECT 1 FROM information_schema.schemata WHERE schema_name = p_schema_name
  );
END;
$$ LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public;

-- Function to create a new schema
CREATE OR REPLACE FUNCTION create_new_schema(p_schema_name text)
RETURNS void AS $$
BEGIN
    IF NOT get_schema_exists(p_schema_name) THEN
        EXECUTE 'CREATE SCHEMA "' || quote_ident(p_schema_name) || '"';
    END IF;
END;
$$ LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public;

-- Function to execute SQL commands
CREATE OR REPLACE FUNCTION sql(command text, schema_name text DEFAULT 'public')
RETURNS text AS $$
#variable_conflict use_variable
DECLARE
    result text;
BEGIN
    IF schema_name != 'public' THEN
        EXECUTE 'SET search_path TO "' || quote_ident(schema_name) || '", public';
    END IF;
    EXECUTE command INTO result;
    RETURN result;
EXCEPTION WHEN others THEN
    RETURN SQLERRM;
END;
$$ LANGUAGE plpgsql
SECURITY DEFINER;


CREATE OR REPLACE FUNCTION ensure_schema_exists(schema_name TEXT)
RETURNS BOOLEAN AS $$
BEGIN
    -- Check if the schema exists
    IF NOT EXISTS (SELECT 1 FROM pg_namespace WHERE nspname = schema_name) THEN
        EXECUTE 'CREATE SCHEMA "' || schema_name || '"';
    END IF;
    RETURN TRUE;
EXCEPTION WHEN OTHERS THEN
    RAISE NOTICE 'Error ensuring schema exists: %', SQLERRM;
    RETURN FALSE;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to create user schema and tables with vector support
CREATE OR REPLACE FUNCTION create_vector_indexes(schema_name TEXT)
RETURNS BOOLEAN AS $$
DECLARE
    success BOOLEAN := TRUE;
BEGIN
    -- Create index for interactions.embedding
    BEGIN
        EXECUTE format('
            CREATE INDEX IF NOT EXISTS %s_interactions_embedding_idx
            ON "%s".interactions
            USING ivfflat (embedding vector_cosine_ops)
            WITH (lists = 100);
        ', schema_name, schema_name);
    EXCEPTION WHEN OTHERS THEN
        -- Log the error but continue
        RAISE NOTICE 'Interactions embedding index creation failed: %', SQLERRM;
        success := FALSE;
    END;

    -- Create index for knowledge_base.embedding
    BEGIN
        EXECUTE format('
            CREATE INDEX IF NOT EXISTS %s_knowledge_base_embedding_idx
            ON "%s".knowledge_base
            USING ivfflat (embedding vector_cosine_ops)
            WITH (lists = 100);
        ', schema_name, schema_name);
    EXCEPTION WHEN OTHERS THEN
        -- Log the error but continue
        RAISE NOTICE 'Knowledge base index creation failed: %', SQLERRM;
        success := FALSE;
    END;

    -- Create index for interaction_embeddings.embedding
    BEGIN
        EXECUTE format('
            CREATE INDEX IF NOT EXISTS %s_interaction_embeddings_idx
            ON "%s".interaction_embeddings
            USING ivfflat (embedding vector_cosine_ops)
            WITH (lists = 100);
        ', schema_name, schema_name);
    EXCEPTION WHEN OTHERS THEN
        -- Log the error but continue
        RAISE NOTICE 'Interaction embeddings index creation failed: %', SQLERRM;
        success := FALSE;
    END;

    RETURN success;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to create user schema and tables with vector support
CREATE OR REPLACE FUNCTION create_user_schema_and_tables(schema_name TEXT)
RETURNS BOOLEAN AS $$
DECLARE
    tables_created BOOLEAN := TRUE;
    indexes_created BOOLEAN := TRUE;
BEGIN
    -- Ensure the schema exists
    PERFORM ensure_schema_exists(schema_name);

    -- Create interactions table
    EXECUTE format('
        CREATE TABLE IF NOT EXISTS "%s".interactions (
            interaction_id SERIAL PRIMARY KEY,
            context TEXT,
            question TEXT,
            answer TEXT,
            metadata JSONB,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            embedding vector(2048), -- Embedding directly in interactions table for backward compatibility
            session_id TEXT
        );', schema_name);

    -- Create knowledge_base table with native vector type
    EXECUTE format('
        CREATE TABLE IF NOT EXISTS "%s".knowledge_base (
            id SERIAL PRIMARY KEY,
            content TEXT,
            embedding vector(2048)
        );', schema_name);

    -- Create interaction_embeddings table with foreign key to interactions
    EXECUTE format('
        CREATE TABLE IF NOT EXISTS "%s".interaction_embeddings (
            id SERIAL PRIMARY KEY,
            interaction_id INTEGER REFERENCES "%s".interactions(interaction_id),
            embedding vector(2048),
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            CONSTRAINT unique_interaction UNIQUE(interaction_id)
        );', schema_name, schema_name);

    -- Use the enhanced ensure_vector_indexes function that provides:
    -- 1. HNSW indexes with fallback to IVFFlat
    -- 2. Session metadata index
    -- 3. Table statistics updates
    -- 4. Robust error handling
    indexes_created := ensure_vector_indexes(schema_name);

    RETURN tables_created AND indexes_created;

EXCEPTION WHEN OTHERS THEN
    RAISE NOTICE 'Error creating schema: %', SQLERRM;
    RETURN FALSE;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Updated function to get conversation history with session_id support
CREATE OR REPLACE FUNCTION get_conversation_history(
    p_schema_name TEXT,
    p_session_id TEXT DEFAULT NULL
)
RETURNS TABLE (
    interaction_id INTEGER,
    context TEXT,
    question TEXT,
    answer TEXT,
    metadata JSONB,
    created_at TIMESTAMP,
    session_id TEXT
) AS $$
BEGIN
    IF p_session_id IS NOT NULL THEN
        -- Filter by session_id if provided
        RETURN QUERY EXECUTE format('
            SELECT interaction_id, context, question, answer, metadata, created_at, session_id
            FROM "%s".interactions
            WHERE session_id = %L
            ORDER BY created_at;', p_schema_name, p_session_id);
    ELSE
        -- Return all interactions if no session_id provided
        RETURN QUERY EXECUTE format('
            SELECT interaction_id, context, question, answer, metadata, created_at, session_id
            FROM "%s".interactions
            ORDER BY created_at;', p_schema_name);
    END IF;

    -- If no records are found, return an empty result set
    RETURN QUERY SELECT NULL::INTEGER, NULL::TEXT, NULL::TEXT, NULL::TEXT, NULL::JSONB, NULL::TIMESTAMP, NULL::TEXT
    WHERE NOT FOUND;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to get topic-specific interactions
CREATE OR REPLACE FUNCTION get_topic_interactions(
    p_schema_name TEXT,
    p_topic_name TEXT,
    p_session_id TEXT DEFAULT NULL,
    p_max_limit INTEGER DEFAULT 5
)
RETURNS TABLE (
    interaction_id INTEGER,
    context TEXT,
    question TEXT,
    answer TEXT,

    created_at TIMESTAMP
) AS $$
BEGIN
    IF p_session_id IS NOT NULL THEN
        RETURN QUERY EXECUTE format('
            SELECT interaction_id, context, question, answer, metadata, created_at

    ELSE
        RETURN QUERY EXECUTE format('
            SELECT interaction_id, context, question, answer, metadata, created_at
            FROM "%s".interactions
            WHERE metadata->>''topic'' = %L
            ORDER BY created_at DESC
            LIMIT %s;', p_schema_name, p_topic_name, p_max_limit);
    END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to add an interaction with proper handling of column case
CREATE OR REPLACE FUNCTION add_interaction(
    p_schema_name TEXT,
    p_context TEXT,
    p_question TEXT,
    p_answer TEXT,
    p_metadata JSONB,
    p_session_id TEXT DEFAULT NULL
)
RETURNS INTEGER AS $$
DECLARE
    new_id INTEGER;
BEGIN

    -- Add session_id to metadata if provided
    IF p_session_id IS NOT NULL THEN
        p_metadata = p_metadata || jsonb_build_object('session_id', p_session_id);
    END IF;

    -- Insert with session_id if the column exists
    IF p_session_id IS NOT NULL THEN

        EXECUTE format('
            INSERT INTO "%s".interactions
            (context, question, answer, metadata, session_id)
            VALUES ($1, $2, $3, $4, $5)
            RETURNING "interaction_id";
        ', p_schema_name)
        INTO new_id
        USING p_context, p_question, p_answer, p_metadata, p_session_id;
    ELSE
        -- Insert without session_id column
        EXECUTE format('
            INSERT INTO "%s".interactions
            (context, question, answer, metadata)
            VALUES ($1, $2, $3, $4)
            RETURNING "interaction_id";
        ', p_schema_name)
        INTO new_id
        USING p_context, p_question, p_answer, p_metadata;
    END IF;

    RETURN new_id;
EXCEPTION
    WHEN OTHERS THEN
        RAISE NOTICE 'Error in add_interaction: %', SQLERRM;
        RETURN NULL;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE FUNCTION find_similar_documents(
    p_schema_name TEXT,
    p_embedding TEXT,
    p_limit INT DEFAULT 5,
    p_min_similarity FLOAT DEFAULT 0.7
)
RETURNS TABLE (
    id BIGINT,
    content TEXT,
    embedding TEXT,
    similarity FLOAT
)
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
BEGIN
    RETURN QUERY EXECUTE format('
        SELECT
            id,
            content,
            embedding::text,
            1 - (embedding <=> %s::vector) as similarity
        FROM %I.knowledge_base
        WHERE 1 - (embedding <=> %s::vector) > %L  -- Ensure minimum similarity
        ORDER BY similarity DESC
        LIMIT %s
    ', p_embedding, p_schema_name, p_embedding, p_min_similarity, p_limit);
END;
$$;

-- Function to connect psychological concepts
CREATE OR REPLACE FUNCTION connect_psychological_concepts(
    p_schema_name TEXT,
    p_source_id INTEGER,
    p_target_id INTEGER,
    p_relationship_type TEXT DEFAULT 'related',
    p_strength FLOAT DEFAULT 1.0
)
RETURNS INTEGER AS $$
DECLARE
    new_id INTEGER;
BEGIN
    -- Ensure the table exists
    PERFORM create_concept_connections_table(p_schema_name);

    -- Insert the new connection
    EXECUTE format('
        INSERT INTO "%s".concept_connections
        (source_id, target_id, relationship_type, strength)
        VALUES ($1, $2, $3, $4)
        ON CONFLICT (source_id, target_id)
        DO UPDATE SET
            relationship_type = $3,
            strength = $4
        RETURNING id;
    ', p_schema_name)
    INTO new_id
    USING p_source_id, p_target_id, p_relationship_type, p_strength;

    RETURN new_id;  -- Return the new or updated ID
EXCEPTION WHEN OTHERS THEN
    RAISE NOTICE 'Error connecting concepts: %', SQLERRM;
    RETURN -1;  -- Return -1 to indicate an error
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to efficiently find psychologically similar interactions with pgvector
CREATE OR REPLACE FUNCTION find_psychological_memories(
    p_schema_name TEXT,
    p_embedding TEXT,
    p_session_id TEXT,
    p_limit INT DEFAULT 5,
    p_threshold FLOAT DEFAULT 0.7
)
RETURNS TABLE (
    interaction_id INTEGER,
    question TEXT,
    answer TEXT,
    context TEXT,
    metadata JSONB,
    similarity FLOAT
) AS $$
BEGIN
    -- Create a session-specific GIN index on metadata->>'session_id' if needed
    BEGIN
        EXECUTE format('
            CREATE INDEX IF NOT EXISTS %I_session_idx
            ON %I.interactions ((metadata->>''session_id''));
        ', p_schema_name, p_schema_name);
    EXCEPTION WHEN OTHERS THEN
        RAISE NOTICE 'Session index creation failed: %', SQLERRM;
    END;

    -- Use ORDER BY for vector search (faster than filtering with WHERE)
    RETURN QUERY EXECUTE format('
        SELECT
            i.interaction_id,
            i.question,
            i.answer,
            i.context,
            i.metadata,
            1 - (ie.embedding <=> %L::vector) as similarity
        FROM
            %I.interactions i
        JOIN
            %I.interaction_embeddings ie
        ON
            i.interaction_id = ie.interaction_id
        WHERE
            i.metadata->>''session_id'' = %L
        ORDER BY
            ie.embedding <=> %L::vector
        LIMIT %s;
    ', p_embedding, p_schema_name, p_schema_name, p_session_id, p_embedding, p_limit);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to analyze psychological theme clusters in vector space
CREATE OR REPLACE FUNCTION analyze_theme_clusters(
    p_schema_name TEXT,
    p_session_id TEXT DEFAULT NULL,
    p_min_similarity FLOAT DEFAULT 0.7,
    p_max_clusters INTEGER DEFAULT 5
)
RETURNS TABLE (
    cluster_id INTEGER,
    dominant_theme TEXT,
    avg_similarity FLOAT,
    sample_questions TEXT[]
) AS $$
BEGIN
    -- Add conditional session_id filter
    IF p_session_id IS NOT NULL THEN
        RETURN QUERY EXECUTE format('
            WITH psychological_data AS (
                SELECT
                    interaction_id,
                    question,
                    embedding,
                    metadata->''topic'' AS theme
                FROM "%s".interactions i
                WHERE embedding IS NOT NULL
                AND i.metadata->>''session_id'' = %L
            ),
            -- Calculate similarity matrix between interactions
            similarity_matrix AS (
                SELECT
                    a.interaction_id as id1,
                    b.interaction_id as id2,
                    a.theme,
                    (1 - (a.embedding <=> b.embedding)) AS similarity
                FROM psychological_data a
                JOIN psychological_data b ON a.interaction_id < b.interaction_id
                WHERE (1 - (a.embedding <=> b.embedding)) > %s
            ),
            -- Group by theme and calculate average similarity
            theme_clusters AS (
                SELECT
                    theme,
                    AVG(similarity) AS avg_similarity,
                    COUNT(*) AS cluster_size,
                    ARRAY_AGG(DISTINCT pd.question) FILTER (WHERE pd.question IS NOT NULL) AS sample_questions
                FROM similarity_matrix sm
                JOIN psychological_data pd ON sm.id1 = pd.interaction_id
                GROUP BY theme
                HAVING COUNT(*) > 1
                ORDER BY AVG(similarity) DESC
                LIMIT %s
            )
            SELECT
                ROW_NUMBER() OVER (ORDER BY avg_similarity DESC)::INTEGER AS cluster_id,
                theme::TEXT,
                avg_similarity,
                sample_questions
            FROM theme_clusters
            WHERE theme IS NOT NULL
        ', p_schema_name, p_session_id, p_min_similarity, p_max_clusters);
    ELSE
        RETURN QUERY EXECUTE format('
            WITH psychological_data AS (
                SELECT
                    interaction_id,
                    question,
                    embedding,
                    metadata->''topic'' AS theme
                FROM "%s".interactions
                WHERE embedding IS NOT NULL
            ),
            -- Calculate similarity matrix between interactions
            similarity_matrix AS (
                SELECT
                    a.interaction_id as id1,
                    b.interaction_id as id2,
                    a.theme,
                    (1 - (a.embedding <=> b.embedding)) AS similarity
                FROM psychological_data a
                JOIN psychological_data b ON a.interaction_id < b.interaction_id
                WHERE (1 - (a.embedding <=> b.embedding)) > %s
            ),
            -- Group by theme and calculate average similarity
            theme_clusters AS (
                SELECT
                    theme,
                    AVG(similarity) AS avg_similarity,
                    COUNT(*) AS cluster_size,
                    ARRAY_AGG(DISTINCT pd.question) FILTER (WHERE pd.question IS NOT NULL) AS sample_questions
                FROM similarity_matrix sm
                JOIN psychological_data pd ON sm.id1 = pd.interaction_id
                GROUP BY theme
                HAVING COUNT(*) > 1
                ORDER BY AVG(similarity) DESC
                LIMIT %s
            )
            SELECT
                ROW_NUMBER() OVER (ORDER BY avg_similarity DESC)::INTEGER AS cluster_id,
                theme::TEXT,
                avg_similarity,
                sample_questions
            FROM theme_clusters
            WHERE theme IS NOT NULL
        ', p_schema_name, p_min_similarity, p_max_clusters);
    END IF;
EXCEPTION WHEN OTHERS THEN
    RAISE NOTICE 'Error in analyze_theme_clusters: %', SQLERRM;
    RETURN;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to analyze emotional trajectory in vector space
CREATE OR REPLACE FUNCTION analyze_emotional_vector_trajectory(
    p_schema_name TEXT,
    p_session_id TEXT
)
RETURNS TABLE (
    segment_id INTEGER,
    start_state TEXT,
    end_state TEXT,
    vector_movement FLOAT,
    similarity_to_progress FLOAT
) AS $$
BEGIN
    -- Create a positive emotional progress vector (this is a simplification)
    RETURN QUERY EXECUTE format('
        WITH emotional_states AS (
            SELECT
                interaction_id,
                metadata->''emotional_state'' AS state,
                (metadata->''emotional_intensity'')::FLOAT AS intensity,
                embedding,
                created_at
            FROM "%s".interactions
            WHERE
                metadata->''session_id'' = %L AND
                metadata->''emotional_state'' IS NOT NULL AND
                embedding IS NOT NULL
            ORDER BY created_at
        )
        SELECT
            ROW_NUMBER() OVER (ORDER BY created_at) AS segment_id,
            curr.state AS start_state,
            lead(curr.state) OVER (ORDER BY curr.created_at) AS end_state,
            curr.intensity AS start_intensity,
            lead(curr.intensity) OVER (ORDER BY curr.created_at) AS end_intensity,
            1 - (curr.embedding <=> lead(curr.embedding) OVER (ORDER BY curr.created_at)) AS vector_movement,
            CASE
                WHEN curr.intensity > lead(curr.intensity) OVER (ORDER BY curr.created_at) THEN 1.0
                ELSE 0.5
            END AS similarity_to_progress
        FROM emotional_states curr
        WHERE end_state IS NOT NULL
        ORDER BY segment_id;
    ', p_schema_name, p_session_id);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to find conceptual connections between different psychological concepts
CREATE OR REPLACE FUNCTION find_concept_connections(
    p_schema_name TEXT,
    p_concept_id INTEGER,
    p_threshold FLOAT DEFAULT 0.7
)
RETURNS TABLE (
    source_id INTEGER,
    target_id INTEGER,
    semantic_similarity FLOAT,
    relationship_strength FLOAT
) AS $$
BEGIN
    -- First verify the concept exists
    PERFORM FROM information_schema.tables
    WHERE table_schema = p_schema_name
    AND table_name = 'knowledge_base';

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Knowledge base table not found in schema %', p_schema_name;
    END IF;

    RETURN QUERY EXECUTE format('
        WITH source_embedding AS (
            SELECT id, embedding
            FROM "%s".knowledge_base
            WHERE id = %L
        ),
        -- Find semantically related concepts
        related_concepts AS (
            SELECT
                %L AS source_id,
                target.id AS target_id,
                (1 - (source.embedding <=> target.embedding)) AS semantic_similarity
            FROM source_embedding source
            CROSS JOIN "%s".knowledge_base target
            WHERE target.id != %L
            AND (1 - (source.embedding <=> target.embedding)) > %L
        ),
        -- Check for explicit connections
        explicit_connections AS (
            SELECT
                source_id,
                target_id,
                strength AS relationship_strength
            FROM "%s".concept_connections
            WHERE source_id = %L OR target_id = %L
        )
        -- Combine semantic and explicit connections
        SELECT
            rc.source_id,
            rc.target_id,
            rc.semantic_similarity,
            COALESCE(ec.relationship_strength, 0) AS relationship_strength
        FROM related_concepts rc
        LEFT JOIN explicit_connections ec
            ON (rc.source_id = ec.source_id AND rc.target_id = ec.target_id)
            OR (rc.source_id = ec.target_id AND rc.target_id = ec.source_id)
        ORDER BY
            -- Prioritize concepts with both semantic similarity and explicit connections
            (rc.semantic_similarity + COALESCE(ec.relationship_strength, 0)) DESC;
    ', p_schema_name, p_concept_id, p_concept_id, p_schema_name, p_concept_id, p_threshold, p_schema_name, p_concept_id, p_concept_id);
EXCEPTION WHEN OTHERS THEN
    RAISE NOTICE 'Error in find_concept_connections: %', SQLERRM;
    RETURN;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to find psychological patterns across sessions
CREATE OR REPLACE FUNCTION find_cross_session_patterns(
    p_user_schema TEXT,
    p_session_ids TEXT[]
)
RETURNS TABLE (
    pattern_topic TEXT,
    session_occurrences INTEGER,
    avg_embedding_similarity FLOAT,
    example_questions TEXT[]
) AS $$
DECLARE
    session_id TEXT;
    temp_result RECORD;
BEGIN
    -- Create a temporary table to collect data across sessions
    CREATE TEMP TABLE session_data (
        topic TEXT,
        question TEXT,
        embedding_json TEXT,
        session_id TEXT
    ) ON COMMIT DROP;

    -- Collect data from all specified sessions
    FOREACH session_id IN ARRAY p_session_ids
    LOOP
        BEGIN
            -- Get interaction data from this session
            EXECUTE format('
                INSERT INTO session_data
                SELECT
                    metadata->''topic'' AS topic,
                    question,
                    embedding::text AS embedding_json,
                    %L AS session_id
                FROM "%s".interactions
                WHERE
                    metadata->''topic'' IS NOT NULL AND
                    embedding IS NOT NULL;
            ', session_id, session_id);
        EXCEPTION WHEN OTHERS THEN
            RAISE NOTICE 'Error processing session %: %', session_id, SQLERRM;
            CONTINUE; -- Skip to next session
        END;
    END LOOP;

    -- Find patterns across sessions
    RETURN QUERY
    WITH topic_sessions AS (
        SELECT
            topic,
            COUNT(DISTINCT session_id) AS session_count,
            ARRAY_AGG(DISTINCT question) FILTER (WHERE question IS NOT NULL) AS questions
        FROM session_data
        GROUP BY topic
        HAVING COUNT(DISTINCT session_id) > 1
    ),
    embedding_similarities AS (
        SELECT
            a.topic,
            a.session_id AS session1,
            b.session_id AS session2,
            -- Approximate vector similarity by converting text back to vector
            -- In real implementation, you'd use proper vector operators
            AVG(1 - (a.embedding_json::vector <=> b.embedding_json::vector)) AS similarity
        FROM session_data a
        JOIN session_data b ON
            a.topic = b.topic AND
            a.session_id < b.session_id -- Compare different sessions
        GROUP BY a.topic, a.session_id, b.session_id
    )
    SELECT
        ts.topic::TEXT AS pattern_topic,
        ts.session_count AS session_occurrences,
        AVG(es.similarity) AS avg_embedding_similarity,
        (SELECT ts.questions[1:3]) AS example_questions
    FROM topic_sessions ts
    JOIN embedding_similarities es ON ts.topic = es.topic
    GROUP BY ts.topic, ts.session_count, ts.questions
    ORDER BY session_occurrences DESC, avg_embedding_similarity DESC;
EXCEPTION WHEN OTHERS THEN
    RAISE NOTICE 'Error in find_cross_session_patterns: %', SQLERRM;
    IF EXISTS (SELECT FROM pg_tables WHERE tablename = 'session_data' AND schemaname = 'pg_temp') THEN
        DROP TABLE session_data;
    END IF;
    RETURN;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to ensure optimized vector indexes for session tables
CREATE OR REPLACE FUNCTION ensure_vector_indexes(
    p_schema_name TEXT
)
RETURNS BOOLEAN AS $$
BEGIN
    -- Create index on interactions table with HNSW (faster)
    BEGIN
        EXECUTE format('
            CREATE INDEX IF NOT EXISTS %I_interactions_hnsw_idx
            ON "%I".interactions
            USING hnsw (embedding vector_cosine_ops)
            WITH (ef_construction = 200, m = 16);
        ', p_schema_name, p_schema_name);

                CREATE INDEX IF NOT EXISTS %I_interactions_ivf_idx

            RAISE NOTICE 'Failed to create interactions index: %', SQLERRM;

    END;

    -- Create index on interaction_embeddings table with HNSW
    BEGIN
        EXECUTE format('
            CREATE INDEX IF NOT EXISTS %I_interaction_embeddings_hnsw_idx
            ON "%I".interaction_embeddings
            USING hnsw (embedding vector_cosine_ops)
            WITH (ef_construction = 200, m = 16);
        ', p_schema_name, p_schema_name);
    EXCEPTION WHEN OTHERS THEN
        -- Fall back to IVFFlat if HNSW failed
        BEGIN
            EXECUTE format('
                CREATE INDEX IF NOT EXISTS %I_interaction_embeddings_ivf_idx
                ON "%I".interaction_embeddings
                USING ivfflat (embedding vector_cosine_ops)
                WITH (lists = 100);
            ', p_schema_name, p_schema_name);
        EXCEPTION WHEN OTHERS THEN
            RAISE NOTICE 'Failed to create interaction_embeddings index: %', SQLERRM;
        END;
    END;

    -- Create index on knowledge_base table with HNSW
    BEGIN
        EXECUTE format('
            CREATE INDEX IF NOT EXISTS %I_knowledge_base_hnsw_idx
            ON "%I".knowledge_base
            USING hnsw (embedding vector_cosine_ops)
            WITH (ef_construction = 200, m = 16);
        ', p_schema_name, p_schema_name);
    EXCEPTION WHEN OTHERS THEN
        -- Fall back to IVFFlat if HNSW failed
        BEGIN

-- Function to find psychological patterns in vector space
CREATE OR REPLACE FUNCTION find_psychological_patterns(
    p_schema_name TEXT,
    p_time_window_days INTEGER DEFAULT 30
)
RETURNS TABLE (
    pattern_group TEXT,
    avg_similarity FLOAT,
    interaction_count INTEGER,
    sample_content TEXT[]
) AS $$
BEGIN
    RETURN QUERY EXECUTE format('
        WITH recent_interactions AS (
            SELECT
                id,
                content,
                embedding,
                metadata->''topic'' AS topic,
                created_at
            FROM "%s".interactions
            WHERE created_at > NOW() - interval ''%s days''
        ),
        embedding_clusters AS (
            SELECT
                id,
                content,
                topic,
                AVG(1 - (a.embedding <=> b.embedding)) AS avg_similarity
            FROM recent_interactions a
            JOIN recent_interactions b ON a.id != b.id AND a.topic = b.topic
            GROUP BY a.id, a.content, a.topic
        )
        SELECT
            topic::TEXT AS pattern_group,
            AVG(avg_similarity) AS avg_similarity,
            COUNT(*) AS interaction_count,
            ARRAY_AGG(content) AS sample_content
        FROM embedding_clusters
        GROUP BY topic
        HAVING AVG(avg_similarity) > 0.7
        ORDER BY AVG(avg_similarity) DESC;
    ', p_schema_name, p_time_window_days);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to initialize knowledge base with therapeutic concepts
CREATE OR REPLACE FUNCTION initialize_knowledge_base(p_schema_name TEXT, p_entries JSONB)
RETURNS BOOLEAN AS $$
DECLARE
    entry JSONB;
    content TEXT;
    embedding TEXT;
    success_count INTEGER := 0;
BEGIN
    -- Create schema and tables if they don't exist
    PERFORM create_user_schema_and_tables(p_schema_name);

    -- Check if knowledge base already has entries
    IF (SELECT COUNT(*) FROM information_schema.tables
        WHERE table_schema = p_schema_name AND table_name = 'knowledge_base') > 0
    THEN
        IF (SELECT COUNT(*) FROM (
            SELECT 1 FROM "%s".knowledge_base LIMIT 1
        ) AS tmp) > 0
        THEN
            RETURN TRUE;
        END IF;
    END IF;

    -- Insert each entry
    FOR i IN 0..jsonb_array_length(p_entries)-1 LOOP
        entry := p_entries->i;
        content := entry->>'content';
        embedding := entry->>'embedding';

        BEGIN
            EXECUTE format('
                INSERT INTO "%s".knowledge_base (content, embedding)
                VALUES (%L, %L::vector)
            ', p_schema_name, content, embedding);

            success_count := success_count + 1;
        EXCEPTION WHEN OTHERS THEN
            RAISE NOTICE 'Error inserting knowledge entry: %', SQLERRM;
        END;
    END LOOP;

    -- Also check if we need to create interaction_embeddings for existing interactions
    BEGIN
        -- Create interaction_embeddings entries for any interactions that have embeddings
        EXECUTE format('
            INSERT INTO "%s".interaction_embeddings (interaction_id, embedding)
            SELECT
                interaction_id,
                embedding
            FROM "%s".interactions
            WHERE
                embedding IS NOT NULL AND
                interaction_id NOT IN (
                    SELECT interaction_id FROM "%s".interaction_embeddings
                )
        ', p_schema_name, p_schema_name, p_schema_name);

        -- Update table statistics for better query planning
        EXECUTE format('ANALYZE "%s".knowledge_base;', p_schema_name);
        EXECUTE format('ANALYZE "%s".interactions;', p_schema_name);
        EXECUTE format('ANALYZE "%s".interaction_embeddings;', p_schema_name);
    EXCEPTION WHEN OTHERS THEN
        RAISE NOTICE 'Error syncing interaction embeddings: %', SQLERRM;
    END;

    RETURN success_count > 0;
EXCEPTION WHEN OTHERS THEN
    RAISE NOTICE 'Error initializing knowledge base: %', SQLERRM;
    RETURN FALSE;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- analyze_conversation_topics
CREATE OR REPLACE FUNCTION analyze_conversation_topics(
    p_schema_name TEXT,
    p_session_id TEXT DEFAULT NULL,
    p_min_count INT DEFAULT 1
)
RETURNS TABLE (
    topic TEXT,
    frequency INTEGER
) AS $$
DECLARE
    has_embeddings BOOLEAN;
BEGIN
    -- Check if we have enough embeddings to analyze
    EXECUTE format('
        SELECT EXISTS (
            SELECT 1 FROM %I.interaction_embeddings
            LIMIT 3
        )', p_schema_name) INTO has_embeddings;

    IF NOT has_embeddings THEN
        RETURN QUERY SELECT 'Not enough conversation data for analysis'::TEXT, 0::INTEGER;
        RETURN;
    END IF;

    -- Use pgvector to cluster similar questions
    IF p_session_id IS NOT NULL THEN
        -- Use session_id filter in metadata
        RETURN QUERY EXECUTE format('
            WITH embedding_clusters AS (
                SELECT
                    i.question,
                    ie.embedding,
                    COALESCE(i.metadata->>''topic'', ''general'')::TEXT as base_topic
                FROM %I.interaction_embeddings ie
                JOIN %I.interactions i ON ie.interaction_id = i.interaction_id
                WHERE i.question IS NOT NULL
                  AND LENGTH(i.question) > 10
                  AND i.metadata->>''session_id'' = %L
            ),
            vector_similarity_groups AS (
                SELECT
                    a.base_topic,
                    a.question,
                    -- Use the dedicated function for consistency
                    (SELECT similarity FROM calculate_vector_similarity(a.embedding, b.embedding)) AS similarity
                FROM embedding_clusters a
                JOIN embedding_clusters b ON a.question != b.question
                WHERE (SELECT similarity FROM calculate_vector_similarity(a.embedding, b.embedding)) > 0.7
            ),
            topic_clusters AS (
                SELECT
                    base_topic as topic,
                    COUNT(DISTINCT question) as frequency
                FROM vector_similarity_groups
                GROUP BY base_topic
                HAVING COUNT(DISTINCT question) >= %L
            )
            SELECT
                topic::TEXT,
                frequency::INTEGER
            FROM topic_clusters
            ORDER BY frequency DESC

            -- Add fallback topic if no clusters detected
            UNION ALL
            SELECT
                CASE WHEN NOT EXISTS (SELECT 1 FROM topic_clusters)
                    THEN ''No significant topics identified''::TEXT
                    ELSE NULL
                END as topic,
                0::INTEGER as frequency
            WHERE NOT EXISTS (SELECT 1 FROM topic_clusters)
        ', p_schema_name, p_schema_name, p_session_id, p_min_count);
    ELSE
        -- No session filter - use all interactions
        RETURN QUERY EXECUTE format('
            WITH embedding_clusters AS (
                SELECT
                    i.question,
                    ie.embedding,
                    COALESCE(i.metadata->>''topic'', ''general'')::TEXT as base_topic
                FROM %I.interaction_embeddings ie
                JOIN %I.interactions i ON ie.interaction_id = i.interaction_id
                WHERE i.question IS NOT NULL AND LENGTH(i.question) > 10
            ),
            vector_similarity_groups AS (
                SELECT
                    a.base_topic,
                    a.question,
                    -- Use the dedicated function for consistency
                    (SELECT similarity FROM calculate_vector_similarity(a.embedding, b.embedding)) AS similarity
                FROM embedding_clusters a
                JOIN embedding_clusters b ON a.question != b.question
                WHERE (SELECT similarity FROM calculate_vector_similarity(a.embedding, b.embedding)) > 0.7
            ),
            topic_clusters AS (
                SELECT
                    base_topic as topic,
                    COUNT(DISTINCT question) as frequency
                FROM vector_similarity_groups
                GROUP BY base_topic
                HAVING COUNT(DISTINCT question) >= %L
            )
            SELECT
                topic::TEXT,
                frequency::INTEGER
            FROM topic_clusters
            ORDER BY frequency DESC

            -- Add fallback topic if no clusters detected
            UNION ALL
            SELECT
                CASE WHEN NOT EXISTS (SELECT 1 FROM topic_clusters)
                    THEN ''No significant topics identified''::TEXT
                    ELSE NULL
                END as topic,
                0::INTEGER as frequency
            WHERE NOT EXISTS (SELECT 1 FROM topic_clusters)
        ', p_schema_name, p_schema_name, p_min_count);
    END IF;

EXCEPTION WHEN OTHERS THEN
    RETURN QUERY SELECT ('Error: ' || SQLERRM)::TEXT, 0::INTEGER;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- calculate_vector_similarity function
DROP FUNCTION IF EXISTS calculate_vector_similarity(vector, vector);
CREATE OR REPLACE FUNCTION calculate_vector_similarity(
    vector1 vector,
    vector2 vector
)
RETURNS TABLE (
    distance float,
    similarity float
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        (vector1 <=> vector2)::float AS distance,
        (1 - (vector1 <=> vector2))::float AS similarity;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;



CREATE OR REPLACE FUNCTION get_columns_types(p_table_name text)
RETURNS TABLE ( column_name text,data_type text ) AS
$BODY$
    BEGIN
        RETURN QUERY EXECUTE format($$SELECT column_name::text,data_type::text FROM information_schema.columns WHERE table_name ='$$|| '%I' ||$$';$$,p_table_name);
    END;
$BODY$
LANGUAGE plpgsql;
