1""" 

2# Psy-Supabase: An AI Therapy Assistant 

3 

4This project implements a therapeutic AI assistant using retrieval-augmented generation (RAG) 

5and associative memory to provide mental health support conversations. 

6 

7## Technical Overview 

8 

9This project uses a multi-model architecture to provide comprehensive therapeutic capabilities: 

10 

111. **Microsoft Phi-1.5**: A lightweight yet capable language model with 1.3 billion parameters, 

12 used for primary response generation. This model offers good performance with reasonable 

13 hardware requirements, making it accessible for deployment on consumer hardware. 

14 

152. **Sentence-Transformers (all-MiniLM-L6-v2)**: Powers the associative memory component, 

16 creating 384-dimensional embeddings that connect related psychological concepts and enable 

17 semantic similarity calculations. 

18 

193. **Facebook RoBERTa Hate Speech Detector**: Uses the facebook/roberta-hate-speech-dynabench-r4-target 

20 model to perform content moderation and toxicity detection. This specialized model ensures 

21 conversations remain safe, appropriate, and therapeutic by identifying potentially harmful 

22 content before processing. 

23 

24These models are augmented with: 

25 

261. **Retrieval-Augmented Generation (RAG)**: Enhances responses by retrieving relevant past 

27 interactions and knowledge documents from a Supabase vector database. 

28 

292. **Associative Memory**: Creates connections between related psychological concepts, 

30 allowing the system to draw on related information (e.g., connecting "anxiety" with 

31 "stress management techniques"). 

32 

333. **Dynamic Template Selection**: Chooses appropriate therapeutic response templates based 

34 on detected pain points and conversation context. 

35 

364. **PostgreSQL Vector Search**: Implements semantic similarity search using pgvector 

37 to find relevant knowledge and past interactions. 

38 

39## Key Components 

40 

41- **RAGProcessor**: Orchestrates the retrieval and generation process 

42- **DynamicRAGRetriever**: Handles semantic search with caching 

43- **AssociativeMemory**: Links related psychological concepts 

44- **DatabaseManager**: Manages the Supabase vector database 

45- **Flask API**: Provides REST endpoints for the frontend application 

46 

47## Configuration 

48 

49The system supports various environment variables: 

50- `SUPABASE_URL` and `SUPABASE_KEY`: Required for database access 

51- `INTELLIGENT_PROCESS_ENABLED`: Toggle advanced processing features (default: true) 

52 

53## Hardware Requirements 

54 

55- **CPU Mode**: Works on any modern CPU with 8GB+ RAM 

56- **GPU Mode (Recommended)**: CUDA-compatible GPU with 6GB+ VRAM 

57- **Storage**: Minimum 1GB for model weights and application

191def should_cleanup_memory() -> bool: 

192 """Determine if we should clean up GPU memory based on request count and time.""" 

193 global REQUEST_COUNTER, last_memory_cleanup # pylint: disable=global-statement 

194 

195 REQUEST_COUNTER += 1 

196 current_time = time.time() 

197 time_since_cleanup = current_time - last_memory_cleanup 

198 

199 # Clean up if: 

200 # 1. We've processed enough requests OR 

201 # 2. It's been long enough since last cleanup OR 

202 # 3. Randomly with low probability (to avoid memory fragmentation) 

203 if ( 

204 REQUEST_COUNTER >= CLEANUP_THRESHOLD or time_since_cleanup >= CLEANUP_TIME_THRESHOLD or random.random() < 0.05 

205 ): # 5% chance to clean up 

206 

207 REQUEST_COUNTER = 0 

208 last_memory_cleanup = current_time 

209 return True 

210 

211 return False

221@app.before_request 

222def before_request() -> Optional[Union[Response, Tuple[Response, int]]]: 

223 """Initialize DatabaseManager before each request.""" 

224 # Skip for preflight requests 

225 if request.method == "OPTIONS": 

226 return None # Consistent return value 

227 

228 # Skip for paths that don't need authentication 

229 if request.path in ["/health", "/memory_status", "/free_memory"]: 

230 return None # Consistent return value 

231 

232 user_id = request.headers.get("X-User-ID") 

233 if not user_id: 

234 return jsonify({"error": "User not authenticated"}), 401 

235 

236 # Store user_id in Flask's 'g' object 

237 g.user_id = user_id 

238 

239 if not supabase_url or not supabase_key: 

240 logger.error("Supabase URL or Key not configured for before_request.") 

241 # Return 500 error as the server is misconfigured 

242 return jsonify({"error": "Server configuration error"}), 500 

243 

244 # Initialize DatabaseManager and store in 'g' 

245 g.db_manager = DatabaseManager(supabase_url, supabase_key, g.user_id) 

246 

247 # Create user schema synchronously 

248 try: 

249 schema_created = g.db_manager.create_user_schema_sync() 

250 if not schema_created: 

251 logger.error("Failed to create user schema for user %s", user_id) 

252 return jsonify({"error": "Failed to initialize user session"}), 500 

253 except Exception as e: 

254 logger.error("Exception during schema creation for user %s: %s", user_id, e, exc_info=True) 

255 return jsonify({"error": "Failed to initialize user session due to server error"}), 500 

256 

257 return None # Consistent return value

289@app.route("/chat", methods=["POST"]) 

290def chat() -> Union[Response, Tuple[Response, int]]: 

291 """Handle chat requests with therapeutic responses.""" 

292 try: 

293 data = request.json 

294 if not data or "question" not in data: 

295 return jsonify({"error": "Missing question parameter"}), 400 

296 

297 user_id = g.user_id # request.headers.get('X-User-ID', 'default_user') 

298 question = data["question"] 

299 

300 # Log the incoming request 

301 logger.info("Received chat request from user %s: %s...", user_id, question[:50]) 

302 

303 # Get the model manager instance and then get the generator 

304 model_manager = get_model_manager(TEXT_GENERATING_MODEL, DEVICE) 

305 generator = model_manager.get_generator() 

306 

307 # Create a RAG processor using the retrieved documents 

308 rag_processor = RAGProcessor(g.db_manager, generator, True) 

309 

310 try: 

311 # Generate response 

312 response = rag_processor.generate_response( 

313 user_question=question, 

314 session_id=user_id, 

315 device=DEVICE, 

316 question_id=0, 

317 ) 

318 

319 finally: 

320 # Always ensure we free memory for expensive operations 

321 # Chat is the most memory-intensive operation, so we clean up explicitly 

322 if DEVICE == "cuda": 

323 cleanup_memory() 

324 

325 return jsonify({"response": response}) 

326 except Exception as e: 

327 logger.error("Error in chat endpoint: %s", e) 

328 logger.error(traceback.format_exc()) 

329 return ( 

330 jsonify( 

331 { 

332 "response": "I apologise, but I encountered an error. Could you try expressing your concern in a different way?" 

333 } 

334 ), 

335 500, 

336 )

339@app.route("/optimize_vectors", methods=["POST"]) 

340def optimize_vectors() -> Union[Response, Tuple[Response, int]]: 

341 """Optimize vector operations for the authenticated user.""" 

342 try: 

343 # Get model manager for embedding generation 

344 model_manager = get_model_manager(TEXT_GENERATING_MODEL, DEVICE) 

345 

346 # Use the database manager to optimize vector operations 

347 # First ensure all interactions have embedding column 

348 g.db_manager.add_embedding_column_to_interactions() 

349 

350 # Find interactions without embeddings 

351 interactions = g.db_manager.get_interactions_without_embeddings() 

352 

353 if not interactions: 

354 return jsonify({"message": "No interactions found that need embeddings."}) 

355 

356 total_interactions = len(interactions) 

357 batch_size = min(10, total_interactions) # Process in smaller batches 

358 enriched_count = 0 

359 

360 logger.info("Starting vector optimization: %d interactions to process", total_interactions) 

361 

362 try: 

363 # Process in batches to avoid memory issues 

364 for i in range(0, total_interactions, batch_size): 

365 batch = interactions[i : i + batch_size] 

366 

367 for interaction in batch: 

368 try: 

369 interaction_id = interaction.get("interaction_id") 

370 question = interaction.get("question", "") 

371 answer = interaction.get("answer", "") 

372 

373 # Generate embedding from combined text - variable removed as unused 

374 text_to_embed = f"Question: {question}\nAnswer: {answer}" 

375 embedding = model_manager.generate_embedding(text_to_embed) 

376 

377 if embedding: 

378 # Add embedding to interaction 

379 if g.db_manager.add_embedding_to_interaction(interaction_id, embedding): 

380 enriched_count += 1 

381 except Exception as e: 

382 logger.error("Error enriching interaction: %s", e) 

383 continue 

384 

385 # After each batch, clean up memory 

386 if DEVICE == "cuda": 

387 cleanup_memory() 

388 

389 # Log progress 

390 logger.info("Processed %d/%d interactions", min(i + batch_size, total_interactions), total_interactions) 

391 finally: 

392 # Ensure memory is cleaned up after the operation 

393 if DEVICE == "cuda": 

394 cleanup_memory() 

395 

396 # Ensure vector indexes exist 

397 g.db_manager.ensure_vector_indexes() 

398 

399 return jsonify( 

400 { 

401 "message": ( 

402 f"Vector operations optimized. {enriched_count}/{total_interactions} " 

403 f"interactions enriched with embeddings." 

404 ) 

405 } 

406 ) 

407 except Exception as e: 

408 logger.error("Error in optimize_vectors endpoint: %s", e) 

409 return jsonify({"error": f"An error occurred: {str(e)}"}), 500
