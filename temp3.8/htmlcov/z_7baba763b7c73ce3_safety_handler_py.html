1""" 

2safety_handler.py 

3 

4This module implements the `SafetyHandler` class, which is responsible for detecting and responding to potentially harmful content 

5in user input. It acts as middleware to process queries before they reach the main RAG system, ensuring that harmful or unsafe 

6content is identified and addressed appropriately. 

7 

8Key Features: 

9 

10- **Harmful Content Detection**: Detects suicide-related, self-harm-related, and violence-related content 

11 using regex patterns. Categorizes harmful content into predefined categories with associated severity levels. 

12 

13- **Safety Interventions**: Generates tailored responses for each category of harmful content. 

14 Provides crisis helpline information and encourages users to seek professional help. 

15 

16- **Metadata Generation**: Creates metadata for detected harmful content, including the category, 

17 severity, and matched pattern. Facilitates logging and tracking of safety interventions. 

18 

19Classes: 

20 

21- `SafetyHandler`: The main class that provides methods for detecting harmful content and generating appropriate responses. 

22 

23Dependencies: 

24 

25- `re`: Used for regex-based pattern matching to detect harmful content. 

26- `prismalog.log.ColoredLogger`: Provides enhanced logging for debugging and monitoring. 

27 

28Usage: 

29 

30.. code-block:: python 

31 

32 from psy_supabase.utilities.safety_handler import SafetyHandler 

33 

34 # Initialize the safety handler 

35 safety_handler = SafetyHandler() 

36 

37 # Process user input 

38 user_input = "I feel like I want to end my life." 

39 is_harmful, response, metadata = safety_handler.process_input(user_input) 

40 

41 if is_harmful: 

42 print("Harmful content detected!") 

43 print("Response:", response) 

44 print("Metadata:", metadata) 

45 else: 

46 print("No harmful content detected.")

72 """Initialize SafetyHandler with detection patterns and response templates."""

101 def process_input(self, user_input: str) -> Tuple[bool, Optional[str], Optional[Dict]]: 

102 """ 

103 Process user input to detect harmful content. 

104 

105 Args: 

106 user_input: The text input from the user 

107 

108 Returns: 

109 Tuple containing: 

110 - is_harmful: Boolean indicating if harmful content was detected 

111 - response: Response text if harmful (None otherwise) 

112 - metadata: Dictionary with detection metadata 

113 """ 

114 for category, data in self.categories.items(): 

115 for pattern in data["patterns"]: 

116 if re.search(pattern, user_input): 

117 logger.warning("Detected %s content: '%s'", category, user_input) 

118 

119 # Get appropriate response 

120 response = data["response"]() 

121 

122 # Create metadata for logging and tracking 

123 metadata = { 

124 "detected_category": category, 

125 "severity": data["severity"], 

126 "matched_pattern": pattern, 

127 "response_type": "safety_intervention", 

128 } 

129 

130 return True, response, metadata 

131 

132 # No harmful content detected 

133 return False, None, None
